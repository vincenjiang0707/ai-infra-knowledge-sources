```python
import paddle
import numpy as np
from paddle import _C_ops

paddle.version.show()

paddle.seed(2025)
np.random.seed(2025)

n = 20

param_np = np.random.uniform(-1, 1, n).astype("float32")
grad_np = np.random.uniform(-1, 1, n).astype("float32")
moment1_np = np.random.uniform(-1, 1, n).astype("float32")
moment2_np = np.random.uniform(0, 1, n).astype("float32")
lr_np = np.random.uniform(0, 1, 1).astype("float32")
beta1_pow_np = np.random.uniform(0, 1, 1).astype("float32")
beta2_pow_np = np.random.uniform(0, 1, 1).astype("float32")
master_param_np = np.random.uniform(-1, 1, n).astype("float32")

# scalar value
beta1 = 0.980554
beta2 = 0.927685
epsilon = 3.46988e-05
coeff = 0.156487
lr_ratio = 0.0388589

def call_adamw(device):
    paddle.set_device(device)

    # create paddle tensor from numpy array
    param_t = paddle.to_tensor(param_np)
    grad_t = paddle.to_tensor(grad_np)
    moment1_t = paddle.to_tensor(moment1_np)
    moment2_t = paddle.to_tensor(moment2_np)
    lr_t = paddle.to_tensor(lr_np)
    beta1_pow_t = paddle.to_tensor(beta1_pow_np)
    beta2_pow_t = paddle.to_tensor(beta2_pow_np)
    master_param_t = paddle.to_tensor(master_param_np)

    # adamw op
    _, _, _, _, _, _, _ = _C_ops.adamw_(
        param_t, #param_and_grad[0],
        grad_t, #param_and_grad[1],
        lr_t, # lr,
        moment1_t, # moment1,
        moment2_t, # moment2,
        None, # moment2_max,
        beta1_pow_t, # beta1_pow_acc,
        beta2_pow_t, # beta2_pow_acc,
        master_param_t, # master_weight,
        None, # found_inf,
        beta1, # _beta1,
        beta2, # _beta2,
        epsilon, # self._epsilon,
        lr_ratio, # lr_ratio_,
        coeff, # self._weight_decay,
        True, # with_decay,
        False, # self._lazy_mode,
        1000, # 1000,
        True, # find_master,
        False, # False,
        False, # self._amsgrad,
    )

    # convert back to numpy array
    return moment1_t.numpy(), moment2_t.numpy(), master_param_t.numpy()

def show_diff(name, cpu_value, gpu_value):
    print(f"cpu vs gpu: max diff for {name}: {np.abs(cpu_value - gpu_value).max()}")

moment1_out_cpu, moment2_out_cpu, master_param_out_cpu = call_adamw("cpu")
moment1_out_gpu, moment2_out_gpu, master_param_out_gpu = call_adamw("gpu:0")

show_diff("moment1", moment1_out_cpu, moment1_out_gpu)
show_diff("moment2", moment2_out_cpu, moment2_out_gpu)
show_diff("master_param", master_param_out_cpu, master_param_out_gpu);

print("cpu master param out:", master_param_out_cpu)
print("gpu master param out:", master_param_out_gpu)
```

### pkuzyc · 2025-03-26

这里输入都是 float32类型，没有使用 AMP（混合精度）。master param 在不开启 AMP 的情况下不会使用，adamw 也不会更新 master param。GPU 上执行 adamw 后 master weight 的值发生变化我们再检查一下。
