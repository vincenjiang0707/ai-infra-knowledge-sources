## 正文

### 文档链接&描述 Document Links & Description

paddle.matmul 的官方文档中参数类型为

> x (Tensor) - 输入变量，类型为 Tensor，数据类型为 bfloat16， float16， float32， float64。
[paddle.matmul](https://www.paddlepaddle.org.cn/documentation/docs/zh/api/paddle/matmul_cn.html#matmul)

然而，实际测试却发现了以下矛盾：

```
import paddle

TEST_DTYPES = [
    'bool',
    'uint8',
    'int8',
    'int16',
    'int32',
    'int64',
    'float16',
    'float32',
    'float64',
    'complex64',
    'complex128'
]

def test_paddle_matmul(device='cpu'):
    results = {}
    
    for dtype in TEST_DTYPES:
        try:
            # 创建指定类型的张量
            a = paddle.randint(1, 10, shape=[3, 3])
            b = paddle.randint(1, 10, shape=[3, 3])
            
            # 特殊处理bool类型
            if dtype == 'bool':
                a = a.astype('bool')
                b = b.astype('bool')
            else:
                a = a.astype(dtype)
                b = b.astype(dtype)
            
            # 转换设备
            if device == 'gpu':
                a = a.cuda()
                b = b.cuda()
            
            # 执行矩阵乘法
            c = a.matmul(b)
            results[dtype] = True
            
        except Exception as e:
            results[dtype] = f"Error: {str(e)}"
    
    return results

def print_results(device, results):
    print(f"\n========== {device.upper()} 测试结果 ==========")
    for dtype, res in results.items():
        status = "成功" if res is True else f"失败: {res}"
        print(f"{dtype.ljust(8)} -> {status}")

if __name__ == "__main__":
    # CPU测试
    cpu_results = test_paddle_matmul('cpu')
    print_results('CPU', cpu_results)
    
    # GPU测试（如果可用）
    if paddle.device.is_compiled_with_cuda():
        gpu_results = test_paddle_matmul('gpu')
        print_results('GPU', gpu_results)
    else:
        print("未检测到可用的CUDA环境")
```
