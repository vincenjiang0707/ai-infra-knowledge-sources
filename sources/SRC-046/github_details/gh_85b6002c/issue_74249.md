    seal_text_detection_model_dir = my_seal_text_detection_model_dir,
    seal_text_recognition_model_dir = my_seal_text_recognition_model_dir,
    formula_recognition_model_dir = my_formula_recognition_model_dir,
    device = "cpu",
    enable_mkldnn = False
)

def ppstructure(file_path: str) -> str:
    """file_path: 文件路径"""
    output = pipeline.predict(file_path)
    filedir = os.path.dirname(file_path)
    img_dir = filedir + '/res_img'
    os.mkdir(img_dir)

    markdown_list = []
    markdown_images = []

    for res in output:
        md_info = res.markdown
        markdown_list.append(md_info)
        markdown_images.append(md_info.get("markdown_images", {}))

    markdown_texts = pipeline.concatenate_markdown_pages(markdown_list)
    mkd_file_path = os.path.join(filedir, "res.md

    with open(mkd_file_path, "w", encoding="utf-8 as f:
        f.write(markdown_texts)

    for item in markdown_images:
        if item:
            for path, image in item.items():
                img_file_path = os.path.join(img_dir, path)
                image.save(img_file_path)

    return markdown_texts

### Bobholamovic · 2025-07-29

你好，这是Paddle框架的一个已知bug导致的。如果你的设备支持MKL-DNN的话，可以通过设置`enable_mkldnn=True`绕过这个问题；如果不支持，可以考虑调整检测模型缩放策略（例如设置`text_det_limit_type`为`"max"`、`text_det_limit_side_len`为`960`），避免传入过大的图像，或者将文本检测模型更换为`PP-OCRv5_mobile_det`，不过这些做法将会影响算法精度效果，可能需要权衡。
