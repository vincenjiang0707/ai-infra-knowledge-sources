## 📌 Description

Fix SM90 FP8 MoE dispatch using rows per physical expert instead of global input
tokens. Configure the estimate after worst-case workspace sizing:

`max(1, ceil(tokens * top_k / (local_experts * ep_size)))`

Keep the existing workspace virtual signatures, base interface, member layout,
allocation logic and kernels unchanged. The BF16×FP8 MoE factory retains its
concrete runner type and sets the existing estimate through a non-virtual setter.
Standalone GEMM workspace behavior is unchanged.

### Source-built H20 EP8 evidence

8×H20 (SM90,78 SM/GPU), FlashInfer0.6.18.post1, PyTorch2.13.0+cu130.
Container digest:
`sha256:8a69ffad015f138d7170c4ddc429e230a3bc1c1719f67e14324749df200a4b90`.
Main geometry: hidden5120/intermediate2304,400 physical experts,50/rank,top-k6.
Draft geometry:128 physical experts,16/rank,top-k3.

The non-virtual implementation was rebuilt and rerun against the saved source
baseline. Both are isolated installed-version backports, not exact upstream
builds. The baseline retains the previous EP argument plumbing but restores
the original dispatch estimate; it is an ablation,not pristine upstream.

| Batch / routing | Baseline ms | Patched ms | Speedup |
|---|---:|---:|---:|
| 288 / uniform | 1.808480 | 0.564064 | 3.21x |
| 192 / uniform | 1.796432 | 0.554240 | 3.24x |
| 2048 / uniform | 1.904528 | 0.720464 | 2.64x |
| 288 /75% routed to six experts | 2.020800 | 1.018144 | 1.98x |
| 288 /six active experts,seven ranks empty | 0.791280 | 0.751328 | 1.05x |
| 1 /uniform | 0.114720 | 0.115152 | ~1.00x |
| Draft288 /uniform | 0.633360 | 0.217456 | 2.91x |

64 rank/case outputs were identical, including two draft cases; reduced outputs
also passed. Full local MoE API timing, not communication-inclusive or model TPS.
Five warmups,24 CUDA Graph/event samples,64MiB L2 flush outside timing,
median of per-sample rank maxima. CUPTI unavailable; clocks unlocked.
Baseline is the same source build with only the estimate restored to input M.

```python
import json
import os
from pathlib import Path
import statistics

import torch
from flashinfer.jit import env as jit_env

assert not os.environ.get('LD_PRELOAD'), 'Source validation must not use interposition'
jit_env.FLASHINFER_AOT_DIR = Path(__file__).parent / os.environ.get('PROBE_MODE', 'baseline')
from flashinfer.fused_moe import cutlass_fused_moe, cutlass_fused_moe_workspace_size

rank = int(os.environ.get('LOCAL_RANK', '0'))
world = int(os.environ.get('WORLD_SIZE', '1'))
assert world == 8, 'This benchmark requires eight real EP ranks'
torch.cuda.set_device(rank)
torch.manual_seed(20260926 + rank)
mode = os.environ.get('PROBE_MODE', 'baseline')
if world > 1:
    torch.distributed.init_process_group('nccl', device_id=torch.device('cuda',rank))
draft = os.environ.get('PROBE_DRAFT') == '1'
experts, hidden, intermediate = (16 if draft else 50), 5120, 2304
total_experts, topk = experts * 8, (3 if draft else 6)
tag = 'draft-' if draft else ''
fp8 = torch.float8_e4m3fn
w1 = torch.empty(experts, intermediate * 2, hidden, device='cuda', dtype=fp8)
w2 = torch.empty(experts, hidden, intermediate, device='cuda', dtype=fp8)
for i in range(experts):
    w1[i].copy_(torch.randn(intermediate * 2, hidden, device='cuda').to(fp8))
    w2[i].copy_(torch.randn(hidden, intermediate, device='cuda').to(fp8))
s1 = torch.rand(experts, intermediate * 2 // 128, hidden // 128, device='cuda') * .015 + .005
s2 = torch.rand(experts, hidden // 128, intermediate // 128, device='cuda') * .015 + .005
flush = torch.empty(64 * 1024 * 1024, dtype=torch.uint8, device='cuda')
results = []

cases = [(288, 'uniform'), (1, 'empty')] if draft else [(2048, 'uniform'), (192, 'uniform'), (288, 'uniform'), (1, 'uniform'), (288, 'skew'), (288, 'empty')]
for m, routing in cases:
    # Inputs, routing and RNG state are identical for baseline and patched runs.
    generator = torch.Generator(device='cuda').manual_seed(1234 + m)
    x = torch.randn(m, hidden, device='cuda', dtype=torch.bfloat16, generator=generator) * .1
    ids = (torch.arange(m * topk, device='cuda', dtype=torch.int32) % total_experts).view(m, topk)
    if routing == 'skew':
        ids[:m * 3 // 4] = torch.arange(topk, device='cuda', dtype=torch.int32)
    elif routing == 'empty':
        ids[:] = torch.arange(topk, device='cuda', dtype=torch.int32)
    weights = torch.full((m, topk), 1/topk, device='cuda')
    out = torch.empty_like(x)
    workspace = torch.empty(cutlass_fused_moe_workspace_size(m, hidden, intermediate,
        total_experts, topk, x_dtype=x.dtype, weight_dtype=fp8, ep_size=8, ep_rank=rank,
        use_deepseek_fp8_block_scale=True), device='cuda', dtype=torch.uint8)

    def run():
        cutlass_fused_moe(input=x, token_selected_experts=ids, token_final_scales=weights,
            fc1_expert_weights=w1, fc2_expert_weights=w2, quant_scales=[s1,s2],
            output_dtype=torch.bfloat16, output=out, ep_size=8, ep_rank=rank,
            use_deepseek_fp8_block_scale=True, workspace_buffer=workspace,
            tune_max_num_tokens=2048)

    for _ in range(5):
        run()
    torch.cuda.synchronize()
    assert torch.isfinite(out).all()
    combined = out.float().clone()
    if world > 1:
        torch.distributed.all_reduce(combined)
    combined_path = Path(__file__).with_name(f'{tag}combined-m{m}-{routing}-world{world}.pt')
    if rank == 0:
        if mode == 'baseline':
            torch.save(combined.cpu(), combined_path)
        else:
            combined_ref = torch.load(combined_path, weights_only=True).to('cuda')
            torch.testing.assert_close(combined, combined_ref, rtol=.005, atol=1e-5)
    out_path = Path(__file__).with_name(f'{tag}baseline-rank{rank}-m{m}-{routing}.pt')
    error = None
    if mode == 'baseline':
        torch.save(out.cpu(), out_path)
    else:
        reference = torch.load(out_path, weights_only=True).to('cuda')
        error = (out.float()-reference.float()).abs().max().item()
        rel = ((out.float()-reference.float()).norm() / reference.float().norm().clamp_min(1e-12)).item()
        assert rel < .005, (rank,m,routing,rel,error)
    graph = torch.cuda.CUDAGraph()
    with torch.cuda.graph(graph):
        run()
    samples = []
    for _ in range(24):
        flush.zero_()
        if world > 1:
            torch.distributed.barrier()
        start, end = torch.cuda.Event(enable_timing=True), torch.cuda.Event(enable_timing=True)
        start.record()
        graph.replay()
        end.record()
        end.synchronize()
        elapsed = torch.tensor([start.elapsed_time(end)], device='cuda')
        if world > 1:
            torch.distributed.all_reduce(elapsed, op=torch.distributed.ReduceOp.MAX)
        samples.append(elapsed.item())
    item = dict(mode=mode, rank=rank, world=world, m=m, routing=routing,
        max_rank_median_ms=statistics.median(samples), max_abs=error,
        samples_ms=samples, timing='CUDA Graph/events; rank MAX; cold L2; all-reduce excluded')
    print(json.dumps(item), flush=True)
    results.append(item)
Path(__file__).with_name(f'{tag}{mode}-rank{rank}-world{world}.json').write_text(json.dumps(results,indent=2))
if world > 1:
    torch.distributed.destroy_process_group()
```
