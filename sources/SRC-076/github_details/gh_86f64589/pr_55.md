## 正文

Most of the repo is LLM Provider agnostic. **`subgraph_extractor`** is one of the exceptions in that it hardcodes usage of OpenAI directly via `EventAdapter` and indirectly via `Orchestrator` . 

**This PR introduces a temporary MUX to enable full pipeline usage directly via an plugboard relay** 
(i.e. no external provider calls required). 
* Subsequent PR's will unify the usage of providers/EventAdapter

_Note: The current implementation of relay does not log the streamed results and only preserves the _iteration_k.json_ at the end of each iteration._

import torch
import torch.nn as nn
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: Fused Linear + Bias + ReLU
# y = ReLU(x @ w.T + b), x:[N,K], w:[M,K], b:[M], y:[N,M] float32
# ----------------------------------------------------------------------------- #

# Autotune configurations for FP32 GEMM
_CONFIGS_LBR = [
    triton.Config({'BLOCK_M': 64,  'BLOCK_N': 64,  'BLOCK_K': 32}, num_warps=4, num_stages=3),
    triton.Config({'BLOCK_M': 128, 'BLOCK_N': 64,  'BLOCK_K': 64}, num_warps=8, num_stages=4),
    triton.Config({'BLOCK_M': 64,  'BLOCK_N': 128, 'BLOCK_K': 64}, num_warps=8, num_stages=4),
    triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 32}, num_warps=8, num_stages=3),
]

@triton.autotune(configs=_CONFIGS_LBR, key=['N','M','K'])
@triton.jit
def _linear_bias_relu_kernel(
    x_ptr, w_ptr, b_ptr, y_ptr,
    N, M, K,
    stride_xn, stride_xk,
    stride_wm, stride_wk,
    stride_yn, stride_ym,
    ACT_RELU: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_n = tl.program_id(axis=0)
    pid_m = tl.program_id(axis=1)
    # tile offsets
    offs_n = pid_n * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_m = pid_m * BLOCK_N + tl.arange(0, BLOCK_N)
    # coalescing hint
    offs_n = tl.max_contiguous(tl.multiple_of(offs_n, BLOCK_M), BLOCK_M)
    offs_m = tl.max_contiguous(tl.multiple_of(offs_m, BLOCK_N), BLOCK_N)
    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # k-loop
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        # load x tile
        x_ptrs = x_ptr + offs_n[:, None] * stride_xn + offs_k[None, :] * stride_xk
        x_mask = (offs_n[:, None] < N) & (offs_k[None, :] < K)
        a = tl.load(x_ptrs, mask=x_mask, other=0.0)
        # load w tile as transposed
        w_ptrs = w_ptr + offs_m[None, :] * stride_wm + offs_k[:, None] * stride_wk
        w_mask = (offs_m[None, :] < M) & (offs_k[:, None] < K)
        bmat = tl.load(w_ptrs, mask=w_mask, other=0.0)
        # matmul
        acc = tl.dot(a, bmat, acc)
    # epilogue: bias and ReLU
    bias = tl.load(b_ptr + offs_m, mask=(offs_m < M), other=0.0)
    acc = acc + bias[None, :]
    if ACT_RELU:
        acc = tl.maximum(acc, 0.0)
    # store
    y_ptrs = y_ptr + offs_n[:, None] * stride_yn + offs_m[None, :] * stride_ym
    y_mask = (offs_n[:, None] < N) & (offs_m[None, :] < M)
    tl.store(y_ptrs, acc.to(y_ptr.dtype.element_ty), mask=y_mask)

def linear_relu_triton(x: torch.Tensor, w: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    # x: [N,K], w:[M,K], b:[M], float32 CUDA
    assert x.is_cuda and w.is_cuda and b.is_cuda
    assert x.dtype == torch.float32 and w.dtype == torch.float32 and b.dtype == torch.float32
    N, Kx = x.shape
    Mw, Kw = w.shape
    assert Kx == Kw and b.numel() == Mw
    M = Mw; K = Kx
    y = torch.empty((N, M), device=x.device, dtype=x.dtype)
    def grid(meta):
        return (triton.cdiv(N, meta['BLOCK_M']), triton.cdiv(M, meta['BLOCK_N']))
    _linear_bias_relu_kernel[grid](
        x, w, b, y,
        N, M, K,
        x.stride(0), x.stride(1),
        w.stride(0), w.stride(1),
        y.stride(0), y.stride(1),
        ACT_RELU=True,
    )
    return y

# ----------------------------------------------------------------------------- #
# Triton kernel: Fused Linear + Bias (no activation)
# y = x @ w.T + b, x:[M,K], w:[N,K], b:[N], y:[M,N]
# ----------------------------------------------------------------------------- #

def _matmul_configs():
    return [
        triton.Config({'BLOCK_SIZE_M': 64,  'BLOCK_SIZE_N': 128, 'BLOCK_SIZE_K': 64}, num_stages=3, num_warps=4),
        triton.Config({'BLOCK_SIZE_M': 128, 'BLOCK_SIZE_N': 128, 'BLOCK_SIZE_K': 64}, num_stages=4, num_warps=8),
        triton.Config({'BLOCK_SIZE_M': 64,  'BLOCK_SIZE_N': 256, 'BLOCK_SIZE_K': 64}, num_stages=4, num_warps=8),
        triton.Config({'BLOCK_SIZE_M': 128, 'BLOCK_SIZE_N': 256, 'BLOCK_SIZE_K': 64}, num_stages=4, num_warps=8),
        triton.Config({'BLOCK_SIZE_M': 128, 'BLOCK_SIZE_N': 128, 'BLOCK_SIZE_K': 128}, num_stages=3, num_warps=8),
    ]

@triton.autotune(configs=_matmul_configs(), key=['M','N','K'])
@triton.jit
def _linear_kernel(
    x_ptr, w_ptr, b_ptr, y_ptr,
    M, N, K,
    stride_xm, stride_xk,
    stride_wn, stride_wk,
    stride_ym, stride_yn,
    BLOCK_SIZE_M: tl.constexpr,
    BLOCK_SIZE_N: tl.constexpr,
    BLOCK_SIZE_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    start_m = pid_m * BLOCK_SIZE_M
    start_n = pid_n * BLOCK_SIZE_N
    offs_m = start_m + tl.arange(0, BLOCK_SIZE_M)
    offs_n = start_n + tl.arange(0, BLOCK_SIZE_N)
    offs_m_safe = tl.where(offs_m < M, offs_m, 0)
    offs_n_safe = tl.where(offs_n < N, offs_n, 0)
    offs_m_safe = tl.max_contiguous(tl.multiple_of(offs_m_safe, BLOCK_SIZE_M), BLOCK_SIZE_M)
    offs_n_safe = tl.max_contiguous(tl.multiple_of(offs_n_safe, BLOCK_SIZE_N), BLOCK_SIZE_N)
    acc = tl.zeros((BLOCK_SIZE_M, BLOCK_SIZE_N), dtype=tl.float32)
    num_k = tl.cdiv(K, BLOCK_SIZE_K)
    for ki in range(num_k):
        offs_k = ki * BLOCK_SIZE_K + tl.arange(0, BLOCK_SIZE_K)
        x_ptrs = x_ptr + offs_m_safe[:, None] * stride_xm + offs_k[None, :] * stride_xk
        w_ptrs = w_ptr + offs_n_safe[None, :] * stride_wn + offs_k[:, None] * stride_wk
        x_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        w_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        a = tl.load(x_ptrs, mask=x_mask, other=0.0)
        bmat = tl.load(w_ptrs, mask=w_mask, other=0.0)
        acc = tl.dot(a, bmat, acc)
    bias = tl.load(b_ptr + offs_n_safe, mask=(offs_n < N), other=0.0)
    acc = acc + bias[None, :]
    y_ptrs = y_ptr + offs_m_safe[:, None] * stride_ym + offs_n_safe[None, :] * stride_yn
    y_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    out = acc.to(y_ptr.dtype.element_ty)
    tl.store(y_ptrs, out, mask=y_mask)

def linear_triton(x: torch.Tensor, w: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    # x:[M,K], w:[N,K], b:[N], float32 CUDA
    assert x.is_cuda and w.is_cuda and b.is_cuda
    assert x.dtype == w.dtype == b.dtype == torch.float32
    M, Kx = x.shape
    Nw, Kw = w.shape
    assert Kx == Kw and b.numel() == Nw
    y = torch.empty((M, Nw), device=x.device, dtype=x.dtype)
    def grid(meta):
        return (triton.cdiv(M, meta['BLOCK_SIZE_M']), triton.cdiv(Nw, meta['BLOCK_SIZE_N']))
    _linear_kernel[grid](
        x, w, b, y,
        M, Nw, Kx,
        x.stride(0), x.stride(1),
        w.stride(0), w.stride(1),
        y.stride(0), y.stride(1),
    )
    return y

# ----------------------------------------------------------------------------- #
# Top-level wrapper: compute 2 hidden layers with ReLU, then final linear
# ----------------------------------------------------------------------------- #

def kernel_function(x: torch.Tensor,
                    w1: torch.Tensor, b1: torch.Tensor,
                    w2: torch.Tensor, b2: torch.Tensor,
                    w3: torch.Tensor, b3: torch.Tensor) -> torch.Tensor:
    """
    Runs 3 fused Triton kernels:
      y1 = ReLU(x @ w1.T + b1)
      y2 = ReLU(y1 @ w2.T + b2)
      y3 = y2  @ w3.T + b3
    x: [128,16384], w1,w2:[16384,16384], b1,b2:[16384], w3:[8192,16384], b3:[8192]
    All float32 CUDA.
    Returns y3: [128,8192]
    """
    # validations
    assert x.is_cuda and w1.is_cuda and b1.is_cuda
    assert w2.is_cuda and b2.is_cuda and w3.is_cuda and b3.is_cuda
    # first hidden
    y1 = linear_relu_triton(x, w1, b1)
    # second hidden
    y2 = linear_relu_triton(y1, w2, b2)
    # final linear
    y3 = linear_triton(y2, w3, b3)
    return y3

# ----------------------------------------------------------------------------- #
# Self-test: compare Triton vs PyTorch
# ----------------------------------------------------------------------------- #

# Original model definition
class Model(nn.Module):
    def __init__(self, input_size, layer_sizes, output_size):
        super(Model, self).__init__()
        layers = []
        current = input_size
        for sz in layer_sizes:
            layers.append(nn.Linear(current, sz))
            layers.append(nn.ReLU())
            current = sz
        layers.append(nn.Linear(current, output_size))
        self.network = nn.Sequential(*layers)
    def forward(self, x):
        return self.network(x)

# Input/initialization helpers
batch_size = 128
input_size = 16384
layer_sizes = [16384, 16384]
output_size = 8192

def get_inputs():
    return [torch.rand(batch_size, input_size)]

def get_init_inputs():
    return [input_size, layer_sizes, output_size]

def run_tests():
    # build reference model
    model = Model(*get_init_inputs()).cuda().eval()
    # random input
    x_cpu = get_inputs()[0]
    x = x_cpu.cuda()
    # extract weights and biases
    layers = [l for l in model.network if isinstance(l, nn.Linear)]
    w1, b1 = layers[0].weight.data.cuda(), layers[0].bias.data.cuda()
    w2, b2 = layers[1].weight.data.cuda(), layers[1].bias.data.cuda()
    w3, b3 = layers[2].weight.data.cuda(), layers[2].bias.data.cuda()
    # Triton output
    y_triton = kernel_function(x, w1, b1, w2, b2, w3, b3)
    # PyTorch output
    with torch.no_grad():
        y_ref = model(x)
    # compare
    if not torch.allclose(y_triton, y_ref, rtol=1e-3, atol=1e-3):
        diff = (y_triton - y_ref).abs().max().item()
        print(f"FAIL: max abs diff = {diff}")
        exit(1)
    print("PASS")

if __name__ == "__main__":
    run_tests()
