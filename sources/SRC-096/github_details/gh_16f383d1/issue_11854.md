# [Issue #11854] CUDA IPC causes GPU OOM for same-process peers

source: https://github.com/openucx/ucx/issues/11854
state: closed | updated: 2026-08-31T05:20:26Z
labels: Bug

## 正文

### Describe the bug

With UCX 1.21.0, UCP selects CUDA IPC for two endpoints in the same process using the same CUDA GPU. A device-to-managed-memory rendezvous transfer then utilizes fragments for intra-process/intra-GPU copies which can grow unthrottled.

This causes GPU memory to decrease by approximately the fragment allocation amount (512 MiB by default) for every retained context and eventually leads to out-of-memory errors. The memory utilization with the same application stays flat with UCX 1.20.0, when CUDA IPC is excluded, or when rendezvous fragments are forced to host memory.

CUDA IPC does not appear useful for this same-process/same-GPU case. The CUDA IPC handle cache also keeps registrations alive, which makes the memory pressure persistent while contexts remain active.

### Steps to Reproduce

Build the standalone reproducer included below. The program creates a loopback UCP client/server pair in one process, uses one visible CUDA GPU, sends from CUDA device memory to a CUDA managed-memory receive buffer, and deliberately retains all contexts.

<details><summary>Standalone reproducer: ucx_same_process_cuda_ipc_oom.c</summary>

```c
// SPDX-FileCopyrightText: Copyright (c) 2026, NVIDIA CORPORATION & AFFILIATES.
// SPDX-License-Identifier: BSD-3-Clause
//
// Demonstrates CUDA IPC registration of UCX rendezvous fragments when both
// peers run in one process and use the same CUDA device.

#include <cuda_runtime_api.h>
#include <ucp/api/ucp.h>

#include <arpa/inet.h>
#include <getopt.h>
#include <inttypes.h>
#include <netinet/in.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define TAG_BASE ((ucp_tag_t)0xc0010000)

typedef struct {
    int          complete;
    ucs_status_t status;
} request_state_t;

typedef struct {
    ucp_conn_request_h conn_request;
} listener_state_t;

typedef struct {
    ucp_context_h context;
    ucp_worker_h  server_worker;
    ucp_worker_h  client_worker;
    ucp_listener_h listener;
} iteration_resources_t;

typedef struct {
    unsigned iterations;
    unsigned warmup;
    unsigned messages;
    size_t   bytes;
    size_t   min_free_bytes;
} options_t;

static void die_status(const char *what, ucs_status_t status)
{
    fprintf(stderr, "%s: %s\n", what, ucs_status_string(status));
    exit(EXIT_FAILURE);
}

static void die_cuda(const char *what, cudaError_t status)
{
    if (status != cudaSuccess) {
        fprintf(stderr, "%s: %s\n", what, cudaGetErrorString(status));
        exit(EXIT_FAILURE);
    }
}

static void send_cb(void *request, ucs_status_t status, void *user_data)
{
    request_state_t *state = user_data;
    (void)request;
    state->status   = status;
    state->complete = 1;
}

static void recv_cb(void *request, ucs_status_t status,
                    const ucp_tag_recv_info_t *info, void *user_data)
{
    request_state_t *state = user_data;
    (void)request;
    (void)info;
    state->status   = status;
    state->complete = 1;
}

static void connection_cb(ucp_conn_request_h conn_request, void *arg)
{
    listener_state_t *state = arg;
    if (state->conn_request == NULL) {
        state->conn_request = conn_request;
    } else {
        fprintf(stderr, "unexpected second connection request\n");
    }
}

static void progress(ucp_worker_h server, ucp_worker_h client)
{
    ucp_worker_progress(server);
    ucp_worker_progress(client);
}

static void wait_request(ucp_worker_h server, ucp_worker_h client, void *request,
                         request_state_t *state, const char *what)
{
    ucs_status_t status;

    if (UCS_PTR_IS_ERR(request)) {
        die_status(what, UCS_PTR_STATUS(request));
    }
    if (!UCS_PTR_IS_PTR(request)) {
        return;
    }

    while (!state->complete) {
        progress(server, client);
    }
    status = ucp_request_check_status(request);
    ucp_request_free(request);
    if (status != UCS_OK) {
        die_status(what, status);
    }
    if (state->status != UCS_OK) {
        die_status(what, state->status);
    }
}

static void print_free_memory(unsigned iteration)
{
    size_t free_bytes, total_bytes;

    die_cuda("cudaMemGetInfo", cudaMemGetInfo(&free_bytes, &total_bytes));
    printf("iteration=%u cuda_free_mib=%zu cuda_total_mib=%zu\n", iteration,
           free_bytes / (1024 * 1024), total_bytes / (1024 * 1024));
    fflush(stdout);
}

static unsigned long parse_ulong(const char *text, const char *option)
{
    char *end;
    unsigned long value = strtoul(text, &end, 0);

    if ((*text == '\0') || (*end != '\0')) {
        fprintf(stderr, "invalid %s value: %s\n", option, text);
        exit(EXIT_FAILURE);
    }
    return value;
}

static void usage(const char *program)
{
    printf("Usage: %s [options]\n\n"
           "  --iterations N    Context create/destroy cycles (default: 64)\n"
           "  --warmup N        Cycles before memory reporting (default: 1)\n"
           "  --messages N      Concurrent GPU tag messages per cycle (default: 8)\n"
           "  --bytes N         Bytes per GPU message (default: 524288)\n"
           "  --min-free-mib N  Fail once free GPU memory is below N MiB (default: 0)\n"
           "  --help            Show this message\n",
           program);
}

static options_t parse_options(int argc, char **argv)
{
    options_t options = {.iterations = 64, .warmup = 1, .messages = 8,
                         .bytes = 512 * 1024, .min_free_bytes = 0};
    static const struct option long_options[] = {
        {"iterations", required_argument, NULL, 'i'},
        {"warmup", required_argument, NULL, 'w'},
        {"messages", required_argument, NULL, 'm'},
        {"bytes", required_argument, NULL, 'b'},
        {"min-free-mib", required_argument, NULL, 'f'},
        {"help", no_argument, NULL, 'h'},
        {NULL, 0, NULL, 0}};
    int option;

    while ((option = getopt_long(argc, argv, "i:w:m:b:f:h", long_options, NULL)) != -1) {
        switch (option) {
        case 'i': options.iterations = (unsigned)parse_ulong(optarg, "--iterations"); break;
        case 'w': options.warmup = (unsigned)parse_ulong(optarg, "--warmup"); break;
        case 'm': options.messages = (unsigned)parse_ulong(optarg, "--messages"); break;
        case 'b': options.bytes = (size_t)parse_ulong(optarg, "--bytes"); break;
        case 'f': options.min_free_bytes = (size_t)parse_ulong(optarg, "--min-free-mib") * 1024 * 1024; break;
        case 'h': usage(argv[0]); exit(EXIT_SUCCESS);
        default: usage(argv[0]); exit(EXIT_FAILURE);
        }
    }
    if ((options.iterations == 0) || (options.messages == 0) || (options.bytes == 0)) {
        fprintf(stderr, "iterations, messages, and bytes must be positive\n");
        exit(EXIT_FAILURE);
    }
    return options;
}

static iteration_resources_t run_iteration(const options_t *options)
{
    ucp_params_t context_params;
    ucp_config_t *config;
    ucp_context_h context;
    ucp_worker_params_t worker_params;
    ucp_worker_h server_worker, client_worker;
    ucp_listener_params_t listener_params;
    ucp_listener_attr_t listener_attr;
    listener_state_t listener_state = {0};
    ucp_listener_h listener;
    ucp_ep_params_t ep_params;
    ucp_ep_h client_ep, server_ep;
    struct sockaddr_in listen_addr;
    ucp_request_param_t send_param, recv_param;
    ucp_tag_recv_info_t recv_info;
    ucp_tag_message_h recv_message;
    void **send_buffers, **recv_buffers, **send_requests, **recv_requests;
    request_state_t *send_states, *recv_states;
    unsigned i;
    ucs_status_t status;

    status = ucp_config_read(NULL, NULL, &config);
    if (status != UCS_OK) die_status("ucp_config_read", status);

    memset(&context_params, 0, sizeof(context_params));
    context_params.field_mask = UCP_PARAM_FIELD_FEATURES;
    context_params.features   = UCP_FEATURE_TAG;
    status = ucp_init(&context_params, config, &context);
    ucp_config_release(config);
    if (status != UCS_OK) die_status("ucp_init", status);

    memset(&worker_params, 0, sizeof(worker_params));
    worker_params.field_mask  = UCP_WORKER_PARAM_FIELD_THREAD_MODE;
    worker_params.thread_mode = UCS_THREAD_MODE_SINGLE;
    status = ucp_worker_create(context, &worker_params, &server_worker);
    if (status != UCS_OK) die_status("ucp_worker_create(server)", status);
    status = ucp_worker_create(context, &worker_params, &client_worker);
    if (status != UCS_OK) die_status("ucp_worker_create(client)", status);

    memset(&listen_addr, 0, sizeof(listen_addr));
    listen_addr.sin_family      = AF_INET;
    listen_addr.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
    listen_addr.sin_port        = 0;
    memset(&listener_params, 0, sizeof(listener_params));
    listener_params.field_mask      = UCP_LISTENER_PARAM_FIELD_SOCK_ADDR |
                                      UCP_LISTENER_PARAM_FIELD_CONN_HANDLER;
    listener_params.sockaddr.addr   = (const struct sockaddr *)&listen_addr;
    listener_params.sockaddr.addrlen = sizeof(listen_addr);
    listener_params.conn_handler.cb = connection_cb;
    listener_params.conn_handler.arg = &listener_state;
    status = ucp_listener_create(server_worker, &listener_params, &listener);
    if (status != UCS_OK) die_status("ucp_listener_create", status);

    memset(&listener_attr, 0, sizeof(listener_attr));
    listener_attr.field_mask = UCP_LISTENER_ATTR_FIELD_SOCKADDR;
    status = ucp_listener_query(listener, &listener_attr);
    if (status != UCS_OK) die_status("ucp_listener_query", status);

    memset(&ep_params, 0, sizeof(ep_params));
    ep_params.field_mask      = UCP_EP_PARAM_FIELD_FLAGS |
                                UCP_EP_PARAM_FIELD_SOCK_ADDR;
    ep_params.flags           = UCP_EP_PARAMS_FLAGS_CLIENT_SERVER;
    ep_params.sockaddr.addr   = (const struct sockaddr *)&listener_attr.sockaddr;
    ep_params.sockaddr.addrlen = sizeof(struct sockaddr_in);
    status = ucp_ep_create(client_worker, &ep_params, &client_ep);
    if (status != UCS_OK) die_status("ucp_ep_create(client)", status);
    while (listener_state.conn_request == NULL) progress(server_worker, client_worker);

    memset(&ep_params, 0, sizeof(ep_params));
    ep_params.field_mask   = UCP_EP_PARAM_FIELD_CONN_REQUEST;
    ep_params.conn_request = listener_state.conn_request;
    status = ucp_ep_create(server_worker, &ep_params, &server_ep);
    if (status != UCS_OK) die_status("ucp_ep_create(server)", status);

    send_buffers  = calloc(options->messages, sizeof(*send_buffers));
    recv_buffers  = calloc(options->messages, sizeof(*recv_buffers));
    send_requests = calloc(options->messages, sizeof(*send_requests));
    recv_requests = calloc(options->messages, sizeof(*recv_requests));
    send_states   = calloc(options->messages, sizeof(*send_states));
    recv_states   = calloc(options->messages, sizeof(*recv_states));
    if (!send_buffers || !recv_buffers || !send_requests || !recv_requests ||
        !send_states || !recv_states) {
        fprintf(stderr, "host allocation failed\n");
        exit(EXIT_FAILURE);
    }

    for (i = 0; i < options->messages; ++i) {
        die_cuda("cudaMalloc(send)", cudaMalloc(&send_buffers[i], options->bytes));
    }

    memset(&recv_param, 0, sizeof(recv_param));
    recv_param.op_attr_mask = UCP_OP_ATTR_FIELD_CALLBACK |
                              UCP_OP_ATTR_FIELD_USER_DATA |
                              UCP_OP_ATTR_FIELD_DATATYPE;
    recv_param.datatype     = ucp_dt_make_contig(1);
    recv_param.cb.recv      = recv_cb;
    memset(&send_param, 0, sizeof(send_param));
    send_param.op_attr_mask = UCP_OP_ATTR_FIELD_CALLBACK |
                              UCP_OP_ATTR_FIELD_USER_DATA |
                              UCP_OP_ATTR_FIELD_DATATYPE;
    send_param.datatype     = ucp_dt_make_contig(1);
    send_param.cb.send      = send_cb;
    for (i = 0; i < options->messages; ++i) {
        send_states[i].status = UCS_OK;
        send_param.user_data  = &send_states[i];
        send_requests[i] = ucp_tag_send_nbx(client_ep, send_buffers[i], options->bytes,
                                             TAG_BASE + i, &send_param);
    }
    for (i = 0; i < options->messages; ++i) {
        do {
            recv_message = ucp_tag_probe_nb(server_worker, TAG_BASE + i, UINT64_MAX,
                                             1, &recv_info);
            if (recv_message == NULL) progress(server_worker, client_worker);
        } while (recv_message == NULL);
        if (recv_info.length != options->bytes) {
            fprintf(stderr, "unexpected message length: %zu\n", recv_info.length);
            exit(EXIT_FAILURE);
        }
        /*
         * UCXX's receive buffer is classified as cuda-managed by UCX.  Use a
         * managed destination explicitly so the standalone reproducer selects
         * the same fragmented rendezvous protocol instead of direct CUDA IPC.
         */
        die_cuda("cudaMallocManaged(recv)",
                 cudaMallocManaged(&recv_buffers[i], options->bytes,
                                   cudaMemAttachGlobal));
        recv_states[i].status = UCS_OK;
        recv_param.user_data  = &recv_states[i];
        recv_requests[i] = ucp_tag_msg_recv_nbx(server_worker, recv_buffers[i],
                                                 recv_info.length, recv_message,
                                                 &recv_param);
    }
    for (i = 0; i < options->messages; ++i) {
        wait_request(server_worker, client_worker, send_requests[i], &send_states[i],
                     "ucp_tag_send_nbx");
        wait_request(server_worker, client_worker, recv_requests[i], &recv_states[i],
                     "ucp_tag_recv_nbx");
    }

    die_cuda("cudaDeviceSynchronize", cudaDeviceSynchronize());
    for (i = 0; i < options->messages; ++i) {
        die_cuda("cudaFree(send)", cudaFree(send_buffers[i]));
        die_cuda("cudaFree(recv)", cudaFree(recv_buffers[i]));
    }
    free(send_buffers); free(recv_buffers); free(send_requests); free(recv_requests);
    free(send_states); free(recv_states);

    (void)client_ep;
    (void)server_ep;
    /* Keep the workers alive: their protocol cache owns the fragment pool. */
    return (iteration_resources_t){.context = context,
                                   .server_worker = server_worker,
                                   .client_worker = client_worker,
                                   .listener = listener};
}

int main(int argc, char **argv)
{
    options_t options = parse_options(argc, argv);
    unsigned i;
    int device_count;
    iteration_resources_t *resources;

    printf("UCX same-process CUDA IPC reproducer\n");
    printf("iterations=%u warmup=%u messages=%u bytes=%zu\n", options.iterations,
           options.warmup, options.messages, options.bytes);
    printf("recv_memory=managed\n");
    die_cuda("cudaGetDeviceCount", cudaGetDeviceCount(&device_count));
    if (device_count < 1) {
        fprintf(stderr, "no CUDA device is visible\n");
        return EXIT_FAILURE;
    }
    die_cuda("cudaSetDevice", cudaSetDevice(0));
    resources = calloc(options.iterations, sizeof(*resources));
    if (resources == NULL) {
        fprintf(stderr, "host allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < options.iterations; ++i) {
        resources[i] = run_iteration(&options);
        if (i >= options.warmup) {
            size_t free_bytes, total_bytes;
            print_free_memory(i + 1);
            die_cuda("cudaMemGetInfo", cudaMemGetInfo(&free_bytes, &total_bytes));
            if ((options.min_free_bytes != 0) && (free_bytes < options.min_free_bytes)) {
                fprintf(stderr, "free GPU memory fell below the requested threshold\n");
                return EXIT_FAILURE;
            }
        }
    }
    for (i = 0; i < options.iterations; ++i) {
        ucp_listener_destroy(resources[i].listener);
        ucp_worker_destroy(resources[i].client_worker);
        ucp_worker_destroy(resources[i].server_worker);
        ucp_cleanup(resources[i].context);
    }
    free(resources);
    printf("completed %u iterations\n", options.iterations);
    return EXIT_SUCCESS;
}
```

### Additional information

A candidate fix is to pass local and remote GPU/system-device identities to the UCT reachability check, and reject CUDA IPC only when peers are both in the same process **and** on the same GPU. This preserves the legitimate same-process, multi-GPU CUDA IPC path.
