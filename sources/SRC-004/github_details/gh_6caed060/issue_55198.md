### 🐛 Describe the bug

`BaseModelLoader.load_model` normally runs these steps after loading model weights:

```python
# vllm/model_executor/model_loader/base_loader.py
        self.load_weights(model, model_config)
        if _has_online_quant(model):
            finalize_layerwise_processing(model, model_config)      # :78
        process_weights_after_loading(model, model_config, target_device)  # :80
    return model.eval()
```

`TensorizerLoader.load_model` overrides this method, and neither of its two branches calls `process_weights_after_loading` or `finalize_layerwise_processing`. The vLLM-tensorized branch also returns without calling `.eval()`.

This can leave quantized checkpoints without required FP8/FP4 requantization, scale folding, or other derived state. The vLLM-tensorized branch may also leave the model in training mode, making this a correctness issue rather than only a performance issue.

### Reproduction

```bash
pip install vllm==0.28.0
python reproduce_vllm_tensorizer_skips_post_load.py   # exit 1
```

<summary><code>reproduce_vllm_tensorizer_skips_post_load.py</code> (self-contained, CPU-only)</summary>

```python
"""vllm: TensorizerLoader.load_model never runs the post-load weight
processing step that every other load format gets.

CPU-only, no GPU, no checkpoint, and the `tensorizer` package is NOT
needed — this is a source/call-graph check:
    pip install vllm==0.28.0
    python reproduce_vllm_tensorizer_skips_post_load.py
"""
import ast
import inspect
import sys

from vllm.model_executor.model_loader import base_loader, tensorizer_loader

TARGETS = ("process_weights_after_loading", "finalize_layerwise_processing")


def calls_in(fn) -> set:
    """Names called in fn's body (attribute calls reduced to the attribute)."""
    try:
        src = inspect.getsource(fn)
    except (OSError, TypeError):
        return set()
    tree = ast.parse(inspect.cleandoc(src))
    out = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Call):
            f = node.func
            if isinstance(f, ast.Name):
                out.add(f.id)
            elif isinstance(f, ast.Attribute):
                out.add(f.attr)
    return out


base = calls_in(base_loader.BaseModelLoader.load_model)
tens = calls_in(tensorizer_loader.TensorizerLoader.load_model)
cpu = calls_in(tensorizer_loader.TensorizerLoader._load_model_serialized_cpu)

print("BaseModelLoader.load_model calls:")
for t in TARGETS:
    print(f"  {t:34s} {t in base}")
print(f"  {'eval':34s} {'eval' in base}")

print("\nTensorizerLoader.load_model (overrides it completely) calls:")
for t in TARGETS:
    print(f"  {t:34s} {t in tens}")
print(f"  {'eval':34s} {'eval' in tens}   <- vLLM-tensorized branch returns "
      f"the model without .eval()")

print("\nTensorizerLoader._load_model_serialized_cpu (the other branch) calls:")
for t in TARGETS:
    print(f"  {t:34s} {t in cpu}")
print(f"  {'eval':34s} {'eval' in cpu}")

# The whole package: where is the model-level post-load step invoked from?
import pathlib
root = pathlib.Path(inspect.getfile(base_loader)).parents[2]
sites = []
for p in root.rglob("*.py"):
    try:
        text = p.read_text(errors="ignore")
    except OSError:
        continue
    for i, line in enumerate(text.splitlines(), 1):
        if "process_weights_after_loading(model" in line:
            sites.append(f"{p.relative_to(root)}:{i}")
print("\nmodel-level post-load call sites in the whole package:")
for s in sites:
    print(f"  {s}")

# --- symptom 1: post-load processing skipped on both tensorizer branches ---
skips_post_load = (not (set(TARGETS) & tens)
                   and not (set(TARGETS) & cpu)
                   and bool(set(TARGETS) & base))

# --- symptom 2: the vLLM-tensorized branch also never calls .eval() ------
# Distinct from symptom 1 and separately observable: BaseModelLoader and
# tensorizer's own CPU branch both return model.eval(); load_model's
# vLLM-tensorized branch returns the model in training mode.
returns_training_mode = ("eval" in base) and ("eval" in cpu) and ("eval" not in tens)

print(f"\nsymptom 1 — post-load weight processing skipped : {skips_post_load}")
print(f"symptom 2 — vLLM-tensorized branch returns a model")
print(f"            still in training mode (no .eval())  : {returns_training_mode}")

bug = skips_post_load or returns_training_mode
print("\nBUG PRESENT" if bug else "\nnot reproduced")
sys.exit(1 if bug else 0)
```

</details>

The CPU-only script compares the loader call paths, confirms that both Tensorizer paths skip post-load processing, and checks that the vLLM-tensorized branch does not call `.eval()`.

### Expected behavior

Route `TensorizerLoader` through the shared post-load steps, or run the equivalent processing in both branches. The returned model should also be in evaluation mode.

_This issue was discovered by a fuzzer, and the report was co-authored with an LLM._
