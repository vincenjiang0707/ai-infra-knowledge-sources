### 🐛 Describe the bug

When a graph input is mutated inside a side-stream region, `torch.compile` can hand back
**pre-mutation data**, even though the user wrote the correct stream join. Eager is correct for the same function.

AOTAutograd functionalizes the in-place mutation into a write to a fresh buffer, and
inductor schedules the write-back `copy_(buf, arg)` on the **mutating** stream, launched
*after* the user's join has already executed. At the moment the join runs, the side
stream's queue holds only the functionalized compute, so the join covers that and not the
write-back. Nothing then orders the caller's stream after the write-back.

```python
import torch

N = 1 << 28
TRIALS = 100

def fn(x, small):
    s = torch.Stream(device="cuda")
    with s:
        x.add_(1)
    # Correct, idiomatic join: make the caller's stream wait for the side
    # stream before anyone reads x.
    torch.cuda.current_stream().wait_stream(s)
    # Keep the returned work tiny so the default stream is idle when the
    # observer read is issued.
    return small + 1 
    # return x


def trials(f, x, small, tail, n=TRIALS):
    stale = 0
    for _ in range(n):
        x.zero_()
        torch.cuda.synchronize()
        f(x, small)
        tail.copy_(x[-8:])  # observer, on the caller's (joined) stream
        torch.cuda.synchronize()

        # If stale value is detected
        if tail.min().item() == 0:
            # print(f"Stale read detected: tail = {tail}")
            stale += 1
        else:
            assert (tail == 1).all()
    return stale


def main():
    compiled = torch.compile(fn, backend="inductor", fullgraph=True)
    x = torch.zeros(N, device="cuda")
    small = torch.zeros(8, device="cuda")
    tail = torch.full((8,), float("nan"), device="cuda")
    compiled(x, small)
    torch.cuda.synchronize()

    for label, f in (("eager   ", fn), ("compiled", compiled)):
        print(f"{label}: stale reads {trials(f, x, small, tail)}/{TRIALS}")


if __name__ == "__main__":
    main()
```
