# [Issue #11817] Inconsistent treatment of source location info in cache key

## 正文

### Describe the bug

There are a few related issues in the cache key computation based on source location info:

- Similar to https://github.com/triton-lang/triton/issues/2597, the cached Triton kernels include the filepath in debug info, but that isn't included in the cache key.
- The line number is included in the cache key even if `TRITON_DISABLE_LINE_INFO=1` is set. Now the cache is getting invalidated even if the output actually isn't changing
- `TRITON_DISABLE_LINE_INFO` unset and 0 hash differently. `get_cache_invalidating_env_vars()` reports only variables that are set, so unset and =0 get different cache keys. This isn't specific to `TRITON_DISABLE_LINE_INFO`.

Suggested fix for the first two: include both file path and starting line number in the cache key iff `TRITON_DISABLE_LINE_INFO` is not set. Something like:

```python
if not knobs.compilation.disable_line_info:
    key += str(self.starting_line_number)
    key += str(self.file_name)
```

Obviously this is annoying for cache portability, but so is including the line number and this gives an escape hatch that doesn't exist for line number currently. IMO someone should set `TRITON_DISABLE_LINE_INFO=1` when they don't care about the source location and want a more portable cache.

Not sure about the last one. It seems like this is just how `get_cache_invalidating_env_vars` works and it's less problematic, but you could give the cache-invalidating env variables some notion of default values.

```python
"""Reproducer: Triton's cache key is inconsistent about source-location inputs.

Three independent issues, one script, no external dependencies beyond Triton.

  1. `starting_line_number` is hashed even when TRITON_DISABLE_LINE_INFO=1,
     where it provably cannot affect the artifact.
  2. The source path is NOT hashed, although it reaches codegen the same way
     the line number does, so the same file name at a different path gets a
     cache hit and receives debug info naming the other directory.
  3. get_cache_invalidating_env_vars() reports presence rather than effective
     value, so unset and an explicitly-set default give one kernel two cache
     keys and two identical cache entries.

Issues 1 and 3 reproduce without a GPU. The artifact comparisons need a working
backend and are reported as SKIPPED otherwise.

Each section runs against its own empty TRITON_CACHE_DIR, so the script is
repeatable and never touches your real cache.

    python repro_cache_key_issues.py
"""

import hashlib
import importlib.util
import os
import sys
import tempfile

import triton
from triton import knobs
from triton._C.libtriton import get_cache_invalidating_env_vars

KERNEL = """
import triton
import triton.language as tl


@triton.jit
def kern(p, N, BLOCK: tl.constexpr):
    offs = tl.arange(0, BLOCK)
    mask = offs < N
    v = tl.load(p + offs, mask=mask, other=0.0)
    tl.store(p + offs, v * 2.0, mask=mask)
"""

SIGNATURE = {"p": "*fp32", "N": "i32", "BLOCK": "constexpr"}
_tmpdir = tempfile.TemporaryDirectory()


def fresh_cache(name: str) -> None:
    """Point Triton at an empty cache dir. Read live, so it takes effect at once.

    Without this the sections leak into each other and across runs: the source
    path is not part of the key (issue 2), so a section can be served an entry
    written by another section, or by a previous run from a since-deleted tmpdir.
    """
    os.environ["TRITON_CACHE_DIR"] = os.path.join(_tmpdir.name, "cache", name)


def build(module_name: str, pad_lines: int = 0, directory: str | None = None,
          basename: str | None = None):
    """Write KERNEL to a file, shifted down by pad_lines, import it, return the kernel."""
    directory = directory or _tmpdir.name
    os.makedirs(directory, exist_ok=True)
    path = os.path.join(directory, basename or f"{module_name}.py")
    with open(path, "w") as f:
        f.write("\n" * pad_lines + KERNEL)
    spec = importlib.util.spec_from_file_location(module_name, path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[module_name] = module
    spec.loader.exec_module(module)
    return module.kern


def compile_kernel(fn):
    src = triton.compiler.ASTSource(fn=fn, signature=SIGNATURE, constexprs={"BLOCK": 128})
    return triton.compile(src)


def artifact_digest(compiled) -> str:
    """Hash every compilation artifact, text stages and binary alike."""
    h = hashlib.sha256()
    for stage in sorted(compiled.asm):
        value = compiled.asm[stage]
        h.update(stage.encode())
        h.update(value if isinstance(value, bytes) else value.encode())
    return h.hexdigest()


def asm_text(compiled) -> str:
    return "\n".join(v for v in compiled.asm.values() if isinstance(v, str))


def issue_1() -> bool:
    print("--- 1. line number is hashed even when line info is disabled")
    os.environ["TRITON_DISABLE_LINE_INFO"] = "1"
    assert knobs.compilation.disable_line_info

    fresh_cache("issue_1")
    a = build("iss1_base")
    b = build("iss1_shifted", pad_lines=7)
    keys_differ = a.cache_key != b.cache_key
    print(f"    same kernel text at line {a.starting_line_number} and {b.starting_line_number}")
    print(f"    cache_key differs: {keys_differ}")

    try:
        da, db = artifact_digest(compile_kernel(a)), artifact_digest(compile_kernel(b))
    except Exception as exc:  # no usable backend
        print(f"    artifact comparison SKIPPED ({type(exc).__name__}: {exc})")
        os.environ.pop("TRITON_DISABLE_LINE_INFO", None)
        return keys_differ
    identical = da == db
    print(f"    every artifact identical: {identical}   ({da[:16]} vs {db[:16]})")
    print("    => different keys, bit-for-bit identical output: a pointless recompile"
          if (keys_differ and identical) else "    => not reproduced")
    os.environ.pop("TRITON_DISABLE_LINE_INFO", None)
    return keys_differ and identical


def issue_2() -> bool:
    print("--- 2. source path is not hashed, though it reaches codegen")
    os.environ.pop("TRITON_DISABLE_LINE_INFO", None)
    assert not knobs.compilation.disable_line_info

    # Same file name in two different directories: only the path differs.
    dir_a = os.path.join(_tmpdir.name, "checkout_a")
    dir_b = os.path.join(_tmpdir.name, "checkout_b")
    fresh_cache("issue_2")
    a = build("iss2_a", directory=dir_a, basename="kernel_mod.py")
    b = build("iss2_b", directory=dir_b, basename="kernel_mod.py")
    same_key = a.cache_key == b.cache_key
    print(f"    same kernel_mod.py in {dir_a}")
    print(f"                      and {dir_b}")
    print(f"    cache_key equal: {same_key}   (so the second is a cache hit)")

    try:
        compile_kernel(a)                     # populate the cache under that key
        text_b = asm_text(compile_kernel(b))  # served from a's entry
    except Exception as exc:
        print(f"    artifact inspection SKIPPED ({type(exc).__name__}: {exc})")
        return same_key
    names_a = dir_a in text_b
    names_b = dir_b in text_b
    for line in text_b.splitlines():
        if ".file" in line and "kernel_mod" in line:
            print(f"    .file directive inside B's artifact: {line.strip()}")
            break
    print(f"    B's artifact embeds A's directory: {names_a}")
    print(f"    B's artifact embeds its own:       {names_b}")
    print("    => the absolute path is in the artifact but not in the key"
          if (same_key and names_a and not names_b) else "    => not reproduced")
    return same_key and names_a and not names_b


def issue_3() -> bool:
    print("--- 3. equivalent environments give the same kernel different cache keys")
    var = "TRITON_DISABLE_LINE_INFO"

    def set_var(value):
        if value is None:
            os.environ.pop(var, None)
        else:
            os.environ[var] = value

    set_var(None)
    unset_env = dict(get_cache_invalidating_env_vars())
    set_var("0")
    off_env = dict(get_cache_invalidating_env_vars())
    set_var(None)
    print(f"    reported env vars, unset: {unset_env}")
    print(f"    reported env vars, ='0' : {off_env}")
    print("    both mean the same thing: line info enabled")

    fresh_cache("issue_3")
    kern = build("iss3")
    try:
        set_var(None)
        unset_compiled = compile_kernel(kern)
        set_var("0")
        off_compiled = compile_kernel(kern)
    except Exception as exc:
        print(f"    compile comparison SKIPPED ({type(exc).__name__}: {exc})")
        set_var(None)
        return unset_env != off_env
    finally:
        set_var(None)

    # compiled.hash is the key Triton hands to get_cache_manager(), i.e. the
    # identity of the on-disk cache entry.
    keys_differ = unset_compiled.hash != off_compiled.hash
    identical = artifact_digest(unset_compiled) == artifact_digest(off_compiled)
    print(f"    cache key, var unset: {unset_compiled.hash}")
    print(f"    cache key, var = '0': {off_compiled.hash}")
    print(f"    cache key differs: {keys_differ}")
    print(f"    every artifact identical: {identical}")
    print("    => same kernel, equivalent settings, second cache entry and a recompile"
          if (keys_differ and identical) else "    => not reproduced")
    print("    (applies to every name in CACHE_INVALIDATING_ENV_VARS, not just this one)")
    return keys_differ and identical


if __name__ == "__main__":
    print(f"triton {triton.__version__}\n")
    results = [issue_1(), issue_2(), issue_3()]
    print(f"\nreproduced {sum(results)}/3")
    sys.exit(0 if all(results) else 1)
```
